cmake_minimum_required(VERSION 3.20)
project(nr_backend_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(GLOB SOURCES "${ROOT}/src/*.cpp" "${ROOT}/src/d3d/*.cpp" "${ROOT}/src/bridge/*.cpp")   # the Direct3D adapters compile to nothing off Windows
list(REMOVE_ITEM SOURCES "${ROOT}/src/main.cpp")
add_executable(backend-isolation backend_isolation.cpp ${SOURCES} "${ROOT}/tools/volk/volk.c")
target_include_directories(backend-isolation PRIVATE "${ROOT}/src" "${ROOT}/tools/volk" "${ROOT}/tools/Vulkan-Headers/include")
target_compile_definitions(backend-isolation PRIVATE VK_ENABLE_BETA_EXTENSIONS NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(WIN32)
  target_compile_definitions(backend-isolation PRIVATE VK_USE_PLATFORM_WIN32_KHR WIN32_LEAN_AND_MEAN)
endif()
if(WIN32)
  target_link_libraries(backend-isolation PRIVATE d3d12 d3d11 dxgi d3dcompiler dxguid)
else()
  target_link_libraries(backend-isolation PRIVATE dl)
endif()
enable_testing()
add_test(NAME backend-isolation COMMAND backend-isolation "${ROOT}/build/shaders")
